Basic Nonlinear GLMM example

This example simulates nonlinear psychometric data and then fits models including a Bayesian nonlinear model using brms.

Simulate data

We construct a dataset with stimulus intensity, stimulus type, and a binomial response.

n_animals <- 10
n_levels <- 7
n_types <- 2
replicates <- 20

width_alpha <- 0.8
width_coef <- 2 * log(2 / (1 - width_alpha) - 1)
inflex <- 15
width <- -10
lapse <- 0.12
base <- 0.5

pop_inflex <- 2.00
pop_sd <- 1.00
pop_width <- 3.00
width_sd <- 1.03

type_inflex <- 2.50
type_width <- 3.5
type_animal_sd <- 0.55
lapse_animal_sd <- 0.50
extra_error_sd <- 0.10

xx <- seq(from = 0, to = n_levels - 1, length.out = 1e3)
plot(x = xx,
     y = base + (1 - lapse - base) * plogis(q = width_coef * (xx - pop_inflex) / pop_width),
     xlim = c(0, n_levels - 1),
     ylim = c(0, 1),
     type = 'l',
     col = 'blue',
     xlab = 'Stimulus',
     ylab = 'Proportion correct')
abline(h = c(0, 1, base, 1 - lapse), lty = c(1, 1, 3, 3))

animal <- rep(x = 1:n_animals, times = n_levels)
animal <- sort(animal)
animal <- rep(x = animal, times = n_types)
stimulus <- rep(x = 1:n_levels - 1, times = n_animals)
stimulus <- rep(x = stimulus, times = n_types)
stype <- rep(x = 1:n_types, times = n_animals * n_levels)
stype <- sort(stype)

animal <- rep(x = animal, times = replicates)
stimulus <- rep(x = stimulus, times = replicates)
stype <- rep(x = stype, times = replicates)

base_bias_animal <- rnorm(n = n_animals, mean = pop_inflex, sd = pop_sd)
width_animal <- rnorm(n = n_animals, mean = pop_width, sd = width_sd)
type_animal <- rnorm(n = n_animals, mean = 1.0, sd = type_animal_sd)
lapse_animal <- plogis(rnorm(n = n_animals, mean = qlogis(lapse), sd = lapse_animal_sd))

response_y <- rep(x = NA, length = n_animals * n_levels * n_types * replicates)
for (ii in seq_along(response_y)) {
  response_y[ii] <- base + (1 - lapse_animal[animal[ii]] - base) *
    plogis(q = width_coef * (stimulus[ii] - (base_bias_animal[animal[ii]] + (stype[ii] - 1) * type_inflex)) /
             (width_animal[animal[ii]] + (stype[ii] - 1) * type_width * type_animal[animal[ii]]) +
             rnorm(n = 1, mean = 0, sd = extra_error_sd))
}

correct_incorrect <- rbinom(n = length(response_y), size = 1, prob = response_y)

sim_data <- data.frame(
  correct_incorrect = correct_incorrect,
  animal = LETTERS[animal],
  stimulus = stimulus,
  type = factor(stype, labels = c('alpha', 'beta'))
)

Define the nonlinear model

We specify a nonlinear formula with transformed parameters for baseline, lapse, inflection, and width.

library(brms)
inv_logit <- inv_logit_scaled

formula_nl <- bf(
  correct_incorrect ~ inv_logit(LogitBase) + (1 - inv_logit(LogitLapse) - inv_logit(LogitBase)) *
    inv_logit(4.39 * (stimulus - Inflex) / exp(LogWidth)),
  LogitBase ~ 1,
  LogitLapse ~ type + (1 + type | animal),
  Inflex ~ type + (1 + type | animal),
  LogWidth ~ type + (1 + type | animal),
  family = bernoulli('identity'),
  nl = TRUE
)

formulaNull_nl <- bf(
  correct_incorrect ~ inv_logit(LogitBase) + (1 - inv_logit(LogitLapse) - inv_logit(LogitBase)) * inv_logit(LogitMean),
  LogitBase ~ 1,
  LogitLapse ~ 1 + (1 | animal),
  LogitMean ~ 1 + (1 | animal),
  family = bernoulli('identity'),
  nl = TRUE
)

prior_nl <- get_prior(formula = formula_nl, data = dta)
priorNull_nl <- get_prior(formula = formulaNull_nl, data = dta)

Set custom priors

We replace default priors with more informative priors for the nonlinear parameters.

prior_nl <- within(prior_nl, {
  prior[class %in% 'b' & nlpar %in% 'LogitBase' & coef %in% 'Intercept'] <- 'normal(0,1)'
  prior[class %in% 'b' & nlpar %in% 'LogitLapse' & coef %in% 'Intercept'] <- 'normal(-3,3)'
  prior[class %in% 'b' & nlpar %in% 'LogitLapse' & coef %in% 'typebeta'] <- 'normal(0,3)'
  prior[class %in% 'b' & nlpar %in% 'Inflex' & coef %in% 'Intercept'] <- 'normal(3,3)'
  prior[class %in% 'b' & nlpar %in% 'Inflex' & coef %in% 'typebeta'] <- 'normal(0,3)'
  prior[class %in% 'b' & nlpar %in% 'LogWidth' & coef %in% 'Intercept'] <- 'normal(1,3)'
  prior[class %in% 'b' & nlpar %in% 'LogWidth' & coef %in% 'typebeta'] <- 'normal(0,3)'
  prior
})

priorNull_nl <- within(priorNull_nl, {
  prior[class %in% 'b' & nlpar %in% 'LogitBase' & coef %in% 'Intercept'] <- prior_nl$prior[prior_nl$class %in% 'b' & prior_nl$nlpar %in% 'LogitBase' & prior_nl$coef %in% 'Intercept']
  prior[class %in% 'b' & nlpar %in% 'LogitLapse' & coef %in% 'Intercept'] <- prior_nl$prior[prior_nl$class %in% 'b' & prior_nl$nlpar %in% 'LogitLapse' & prior_nl$coef %in% 'Intercept']
  prior[class %in% 'b' & nlpar %in% 'LogitLapse' & coef %in% 'typebeta'] <- prior_nl$prior[prior_nl$class %in% 'b' & prior_nl$nlpar %in% 'LogitLapse' & prior_nl$coef %in% 'typebeta']
  prior[class %in% 'b' & nlpar %in% 'LogitMean' & coef %in% 'Intercept'] <- 'normal(0,3)'
  prior
})

Fit the model

The Bayesian model is fit in stages: a prior-only dummy run, a short run, and a full run.

# Dummy prior-only fit
dummy_fit <- brm(formula = formula_nl, data = dta, prior = prior_nl, sample_prior = 'only', iter = 300, chains = 4, cores = 4, refresh = 0, backend = 'cmdstanr')
plot(dummy_fit, nvariables = 10, variable = '^b_', regex = TRUE)
plot(dummy_fit, nvariables = 10, variable = '^sd', regex = TRUE)

# Short fit for convergence check
short_fit <- brm(formula = formula_nl, data = dta, prior = prior_nl, iter = 300, chains = 4, cores = 4, refresh = 0, backend = 'cmdstanr')
plot(short_fit, nvariables = 10, variable = '^b_', regex = TRUE)
plot(short_fit, nvariables = 10, variable = '^sd', regex = TRUE)

# Full fit
full_fit <- brm(formula = formula_nl, data = dta, prior = prior_nl, iter = 1000, chains = 4, cores = 4, refresh = 0, backend = 'cmdstanr')
null_fit <- brm(formula = formulaNull_nl, data = dta, prior = priorNull_nl, iter = 1000, chains = 4, cores = 4, refresh = 0, backend = 'cmdstanr')
