Hypothesis testing example

This example demonstrates two-group hypothesis tests using base R functions like t.test and wilcox.test. It also uses the report package for easy reporting of results.

Enter data

We will simulate random data for two experimental conditions, A and B.

dt_A <- rnorm(n = 10, mean = 1.0, sd = 0.3)
fct_A <- rep(x = 'A', times = length(dt_A))

dt_B <- rnorm(n = 10, mean = 1.5, sd = 0.3)
fct_B <- rep(x = 'B', times = length(dt_B))

dt <- data.frame(
  value = c(dt_A, dt_B),
  condition = c(fct_A, fct_B)
)

summary(dt)
##      value         condition        
##  Min.   :0.6043   Length:20

##  Median :1.3087   Mode  :character  
##  Mean   :1.2435

Inspect the data

We will visualise the distribution with a boxplot and raw data points.

boxplot(value ~ condition,
        data = dt,
        ylab = 'body mass (g)',
        xlab = 'condition',
        ylim = range(dt$value),
        pars = list(boxwex = 0.5),
        outline = FALSE,
        col = 'gray95',
        cex.lab = 1.5)

stripchart(value ~ condition,
           data = dt,
           vertical = TRUE,
           add = TRUE,
           method = 'jitter',
           pch = 21,
           bg = 'darkblue',
           col = 'white',
           cex = 1.5)

Perform a Student’s t-test

We ask whether condition predicts value, using a two-sided Student’s t-test with equal variances.

tt <- t.test(
  formula = value ~ condition,
  data = dt,
  alternative = "two.sided",
  var.equal = TRUE
)

print(tt)
## 
##  Two Sample t-test
## 
## data:  value by condition
## t = -3.2324, df = 18, p-value = 0.004621
## alternative hypothesis: true difference in means between group A and group B is not equal to 0
## 95 percent confidence interval:
##  -0.6878178 -0.1459245
## sample estimates:
## mean in group A mean in group B

Welch’s t-test for unequal variances

Now we simulate a dataset where the two groups have different variances and apply Welch’s t-test.

dt_unequal <- within(dt, {
  value[condition %in% 'B'] <- rnorm(n = 10, mean = 1.5, sd = 0.15)
})

summary(dt_unequal)
##      value         condition        
##  Min.   :0.6043   Length:20         
##  1st Qu.:1.1081   Class :character  
##  Median :1.4116   Mode  :character  
##  Mean   :1.2885                     
##  3rd Qu.:1.5827                     
##  Max.   :1.6866
tt_welch <- t.test(
  formula = value ~ condition,
  data = dt_unequal,
  alternative = "two.sided",
  var.equal = FALSE
)

print(tt_welch)

Non-normal distributions and the Mann-Whitney test

For data that may violate the normality assumption, we simulate an asymmetric distribution and compare groups using Welch’s t-test and the Mann-Whitney rank sum test.

dt_un_A <- rnorm(n = 10, mean = 1.0, sd = 0.3) + rpois(n = 10, lambda = 3.4)
fct_un_A <- rep(x = 'A', times = length(dt_un_A))

dt_un_B <- rnorm(n = 10, mean = 3.5, sd = 0.6) + rpois(n = 10, lambda = 3.4)
fct_un_B <- rep(x = 'B', times = length(dt_un_B))

dt_unknown <- data.frame(
  value = c(dt_un_A, dt_un_B),
  condition = c(fct_un_A, fct_un_B)
)

summary(dt_unknown)
##      value        condition        
##  Min.   :1.794   Length:20         
##  1st Qu.:4.212   Class :character  
##  Median :5.770   Mode  :character  
##  Mean   :5.850                     
##  3rd Qu.:7.848                     
##  Max.   :8.740
boxplot(value ~ condition,
        data = dt_unknown,
        ylab = 'body mass (g)',
        xlab = 'condition',
        ylim = range(dt_unknown$value),
        pars = list(boxwex = 0.5),
        outline = FALSE,
        col = 'gray95',
        cex.lab = 1.5)

stripchart(value ~ condition,
           data = dt_unknown,
           vertical = TRUE,
           add = TRUE,
           method = 'jitter',
           pch = 21,
           bg = 'darkblue',
           col = 'white',
           cex = 1.5)

mean_unknown <- aggregate(
  value ~ condition,
  data = dt_unknown,
  FUN = function(x) {
    sum_stat <- mean(x) + 1.96 * c(0, -1, 1) * sd(x) / sqrt(length(x))
    names(sum_stat) <- c('mean', 'lower CI', 'upper CI')
    return(sum_stat)
  }
)

print(mean_unknown)
##   condition value.mean value.lower CI value.upper CI
## 1         A   4.664204       3.396375       5.932033
## 2         B   7.035760       6.023326       8.048195
tt_unknown <- t.test(
  formula = value ~ condition,
  data = dt_unknown,
  alternative = "two.sided",
  var.equal = FALSE
)

print(tt_unknown)
## 
##  Welch Two Sample t-test
## 
## data:  value by condition
## t = -2.8649, df = 17.16, p-value = 0.01066
## alternative hypothesis: true difference in means between group A and group B is not equal to 0
## 95 percent confidence interval:
##  -4.1168030 -0.6263092
## sample estimates:
## mean in group A mean in group B 
##        4.664204        7.035760
mww <- wilcox.test(
  value ~ condition,
  data = dt_unknown,
  alternative = "two.sided"
)

print(mww)
## 
##  Wilcoxon rank sum exact test
## 
## data:  value by condition
## W = 19, p-value = 0.01854
## alternative hypothesis: true location shift is not equal to 0

Rank-transformed visualisation

The plot below shows both the original values and the rank-transformed values for the two conditions.

par(mfrow = c(1, 2))
with(dt_unknown, {
  hist(value[condition %in% 'A'],
       breaks = 1e2,
       col = 'blue',
       xlim = range(value),
       xlab = 'value',
       main = 'original values')
  hist(value[condition %in% 'B'],
       breaks = 1e2,
       col = 'red',
       add = TRUE)
  hist(rank(value)[condition %in% 'A'],
       breaks = 1e2,
       col = 'blue',
       xlim = c(0, max(rank(value))),
       xlab = 'value',
       main = 'rank-transformed')
  hist(rank(value)[condition %in% 'B'],
       breaks = 1e2,
       col = 'red',
       add = TRUE)
})
